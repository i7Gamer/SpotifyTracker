# Test-only dependencies. NOT installed into the Docker image - requirements.txt
# is what the image gets, and .dockerignore keeps the rest of the dev tooling out.
#
# This file exists because pyproject.toml's [tool.pytest.ini_options] sets
# `addopts = "-n auto"`, which makes pytest-xdist mandatory rather than optional:
# without it a bare `pytest` does not run the suite slowly, it refuses to start
# at all ("unrecognized arguments: -n auto", exit 4) - naming a flag the user
# never typed. Nothing declared these, and the README documented no test setup,
# so a fresh clone could not run the tests.
#
# Pinned for the same reason ruff and eslint are pinned in the lint workflow: a
# new pytest major can change collection or deprecate an API and turn CI red on
# a push that touched none of it.
pytest==9.1.1
pytest-xdist==3.8.0

# Used to find untested paths - routes/system.py's three polled status
# endpoints were found this way, at 78% with no test at all (they are at 100%
# now). Not needed to RUN the suite, but declared so `coverage run -m pytest`
# works from a clone.
coverage==7.16.1

# What lets CI measure coverage without a 7-minute serial run. `coverage run`
# alone cannot see pytest-xdist's worker SUBPROCESSES - it needs
# COVERAGE_PROCESS_START plus a coverage.process_startup() hook installed via a
# .pth file, which is machinery to build and get wrong. pytest-cov does exactly
# that, so `pytest --cov -n auto` just works and the measured run stays
# parallel (~2min vs ~7min serial).
#
# Pinned like everything else here: a plugin that changes its collection
# behaviour turns CI red on a push that touched none of it.
pytest-cov==7.1.0

# A CSS selector engine (soupsieve, which beautifulsoup4 pulls in) so the
# page-gap guard can ask what the stylesheet's rule ACTUALLY selects instead of
# grepping for its text. Two positional spellings of that rule had shipped
# wrong, both passing a string-match guard that could not see whom they hit.
# It was already installed here as somebody else's transitive dependency, which
# is exactly why it needed declaring: nothing guaranteed a fresh clone got it.
beautifulsoup4==4.15.0
# Pinned for the same reason, and it already happened: soupsieve 2.10
# answers `.card:has(~ *)` as false for the last of three cards even with a
# section after it, so the page-gap guard failed on an unchanged main
# (CI, 2026-09-24). 2.9.2 and browsers agree the card has a following sibling.
soupsieve==2.9.2
